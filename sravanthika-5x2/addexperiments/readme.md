## PL/SQL CODE
```
DECLARE
    V_ID STUDENT.STUDENT_ID%TYPE := &STUDENT_ID;
    V_NAME STUDENT.NAME%TYPE;
    V_DEPT STUDENT.DEPARTMENT%TYPE;
    V_MARKS STUDENT.MARKS%TYPE;
BEGIN
    SELECT NAME, DEPARTMENT, MARKS
    INTO V_NAME, V_DEPT, V_MARKS
    FROM STUDENT
    WHERE STUDENT_ID = V_ID;

    DBMS_OUTPUT.PUT_LINE('Student ID : ' || V_ID);
    DBMS_OUTPUT.PUT_LINE('Name       : ' || V_NAME);
    DBMS_OUTPUT.PUT_LINE('Department : ' || V_DEPT);
    DBMS_OUTPUT.PUT_LINE('Marks      : ' || V_MARKS);

EXCEPTION
    WHEN NO_DATA_FOUND THEN
        DBMS_OUTPUT.PUT_LINE('Student ID not found.');
END;
/
```

CREATE OR REPLACE FUNCTION ANNUAL_SALARY
(
    P_SALARY NUMBER
)
RETURN NUMBER
IS
BEGIN
    RETURN P_SALARY * 12;
END;
/

````
DECLARE
    V_DEPT VARCHAR2(30) := '&DEPARTMENT';

    CURSOR EMP_CURSOR(P_DEPT VARCHAR2) IS
        SELECT ID, ENAME, DEPARTMENT, DESIGNATION, SALARY
        FROM EMPLOYEE
        WHERE UPPER(DEPARTMENT) = UPPER(P_DEPT);

BEGIN
    DBMS_OUTPUT.PUT_LINE('Employees in Department: ' || V_DEPT);
    DBMS_OUTPUT.PUT_LINE('----------------------------------------');

    FOR EMP IN EMP_CURSOR(V_DEPT)
    LOOP
        DBMS_OUTPUT.PUT_LINE(
            'ID: ' || EMP.ID ||
            ' | Name: ' || EMP.ENAME ||
            ' | Designation: ' || EMP.DESIGNATION ||
            ' | Salary: ' || EMP.SALARY
        );
    END LOOP;
END;
/
```
