## PL/SQL CODE
```
CREATE OR REPLACE FUNCTION COUNT_STUDENTS(
    P_COURSE IN VARCHAR2
)
RETURN NUMBER
IS
    V_COUNT NUMBER;
BEGIN
    SELECT COUNT(*)
    INTO V_COUNT
    FROM STUDENT
    WHERE COURSE = P_COURSE;

    RETURN V_COUNT;
END;
/
```

##PL/SQL CODE
```
CREATE OR REPLACE FUNCTION GET_GRADE(
    P_MARKS IN NUMBER
)
RETURN VARCHAR2
IS
    V_GRADE VARCHAR2(20);
BEGIN
    IF P_MARKS >= 75 THEN
        V_GRADE := 'Distinction';

    ELSIF P_MARKS >= 60 THEN
        V_GRADE := 'First Class';

    ELSIF P_MARKS >= 50 THEN
        V_GRADE := 'Second Class';

    ELSIF P_MARKS >= 35 THEN
        V_GRADE := 'Pass';

    ELSE
        V_GRADE := 'Fail';
    END IF;

    RETURN V_GRADE;
END;
/
```
