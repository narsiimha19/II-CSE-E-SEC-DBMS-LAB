-- 1. Create the table
CREATE TABLE employee (
    emp_id   NUMBER PRIMARY KEY,
    emp_name VARCHAR2(50),
    salary   NUMBER(10,2)
);

-- 2. Create a BEFORE INSERT trigger
CREATE OR REPLACE TRIGGER trg_employee_validate
BEFORE INSERT ON employee
FOR EACH ROW
BEGIN
    -- Validate employee name
    IF :NEW.emp_name IS NULL THEN
        RAISE_APPLICATION_ERROR(-20001,
            'Employee name cannot be NULL');
    END IF;

    -- Validate salary
    IF :NEW.salary <= 0 THEN
        RAISE_APPLICATION_ERROR(-20002,
            'Salary must be greater than zero');
    END IF;
END;
/

-- 3. Insert a valid record
INSERT INTO employee (emp_id, emp_name, salary)
VALUES (101, 'Ravi', 25000);

-- 4. Insert an invalid record
-- This will raise an application error because salary is negative.
INSERT INTO employee (emp_id, emp_name, salary)
VALUES (102, 'Kiran', -5000);

-- 5. Display the table contents
SELECT * FROM employee;

-- Program 2: AFTER Trigger
-- 1. Create the main table
CREATE TABLE employee (
    emp_id NUMBER PRIMARY KEY,
    emp_name VARCHAR2(50),
    salary NUMBER
);

-- 2. Create the audit table
CREATE TABLE employee_audit (
    emp_id NUMBER,
    emp_name VARCHAR2(50),
    salary NUMBER,
    action_type VARCHAR2(20),
    action_date DATE
);

-- 3. Create AFTER INSERT trigger
CREATE OR REPLACE TRIGGER trg_employee_audit
AFTER INSERT ON employee
FOR EACH ROW
BEGIN
    -- Record newly inserted data in audit table
    INSERT INTO employee_audit
        (emp_id, emp_name, salary, action_type, action_date)
    VALUES
        (:NEW.emp_id, :NEW.emp_name, :NEW.salary,
         'INSERT', SYSDATE);
END;
/

-- 4. Insert a new record into the main table
INSERT INTO employee (emp_id, emp_name, salary)
VALUES (101, 'Ravi', 25000);

-- 5. Save the transaction
COMMIT;

-- 6. Display the main table
SELECT * FROM employee;

-- 7. Display the audit table
SELECT * FROM employee_audit;

program 3:

-- Program 3: Row-Level Trigger

-- 1. Create the required table
CREATE TABLE employee (
    emp_id NUMBER PRIMARY KEY,
    emp_name VARCHAR2(50),
    salary NUMBER
);

-- 2. Insert sample records
INSERT INTO employee VALUES (101, 'Ravi', 25000);
INSERT INTO employee VALUES (102, 'Kiran', 30000);

COMMIT;

-- 3. Create BEFORE UPDATE Row-Level Trigger
CREATE OR REPLACE TRIGGER trg_validate_salary
BEFORE UPDATE ON employee
FOR EACH ROW
BEGIN
    -- Compare OLD salary with NEW salary
    IF :NEW.salary < :OLD.salary THEN
        RAISE_APPLICATION_ERROR(
            -20001,
            'Salary cannot be decreased'
        );
    END IF;
END;
/

-- 4. Update a valid record
-- Salary is increased from 25000 to 28000
UPDATE employee
SET salary = 28000
WHERE emp_id = 101;

COMMIT;

-- 5. Display table after valid update
SELECT * FROM employee;

-- 6. Try an invalid update
-- Salary is decreased from 30000 to 25000
UPDATE employee
SET salary = 25000
WHERE emp_id = 102;

-- This update will be rejected by the trigger.

program 4:

-- Program 4: Statement-Level Trigger

-- 1. Create the main table
CREATE TABLE employee (
    emp_id NUMBER PRIMARY KEY,
    emp_name VARCHAR2(50),
    salary NUMBER
);

-- 2. Insert sample records
INSERT INTO employee VALUES (101, 'Ravi', 25000);
INSERT INTO employee VALUES (102, 'Kiran', 30000);
INSERT INTO employee VALUES (103, 'Arun', 28000);

COMMIT;

-- 3. Create the log table
CREATE TABLE delete_log (
    log_id NUMBER GENERATED BY DEFAULT AS IDENTITY,
    action_type VARCHAR2(20),
    action_date DATE
);

-- 4. Create AFTER DELETE Statement-Level Trigger
CREATE OR REPLACE TRIGGER trg_employee_delete
AFTER DELETE ON employee
BEGIN
    INSERT INTO delete_log
        (action_type, action_date)
    VALUES
        ('DELETE', SYSDATE);

    DBMS_OUTPUT.PUT_LINE(
        'DELETE operation completed successfully.'
    );
END;
/

-- 5. Enable output
SET SERVEROUTPUT ON;

-- 6. Delete one or more records
DELETE FROM employee
WHERE emp_id IN (101, 102);

COMMIT;

-- 7. Display remaining records
SELECT * FROM employee;

-- 8. Display the delete log
SELECT * FROM delete_log;

-- Program 5: INSTEAD OF Trigger

-- 1. Create the base table
CREATE TABLE employee (
    emp_id NUMBER PRIMARY KEY,
    emp_name VARCHAR2(50),
    salary NUMBER
);

-- 2. Insert sample records
INSERT INTO employee VALUES (101, 'Ravi', 25000);
INSERT INTO employee VALUES (102, 'Kiran', 30000);

COMMIT;

-- 3. Create a view using the base table
CREATE OR REPLACE VIEW employee_view AS
SELECT emp_id, emp_name, salary
FROM employee;

-- 4. Create INSTEAD OF UPDATE trigger on the view
CREATE OR REPLACE TRIGGER trg_employee_view_update
INSTEAD OF UPDATE ON employee_view
FOR EACH ROW
BEGIN
    -- Update the corresponding record in the base table
    UPDATE employee
    SET emp_name = :NEW.emp_name,
        salary   = :NEW.salary
    WHERE emp_id = :OLD.emp_id;
END;
/

-- 5. Update a record through the view
UPDATE employee_view
SET salary = 35000

WHERE emp_id = 101;

COMMIT;

-- 6. Display the view
SELECT * FROM employee_view;

-- 7. Display the base table
SELECT * FROM employee;
